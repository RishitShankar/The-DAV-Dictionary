    const loginForm = document.getElementById('loginForm');
    const userIdInput = document.getElementById('userIdInput');
    const passwordInput = document.getElementById('passwordInput');
    const togglePasswordBtn = document.getElementById('togglePasswordBtn');
    const feedbackMsg = document.getElementById('feedbackMsg');
    const welcomePanel = document.getElementById('welcomePanel');
    const loggedInUserSpan = document.getElementById('loggedInUserSpan');
    const logoutBtn = document.getElementById('logoutBtn');

    function showFeedback(message) {
      feedbackMsg.textContent = message;
      feedbackMsg.classList.remove('hidden');
    }

    function clearFeedback() {
      feedbackMsg.textContent = '';
      feedbackMsg.classList.add('hidden');
    }

    const passwordStateBadge = document.getElementById('passwordStateBadge');
    const toggleCodeViewBtn = document.getElementById('toggleCodeViewBtn');
    const copyCodeBtn = document.getElementById('copyCodeBtn');
    const codePreviewBox = document.getElementById('codePreviewBox');

    function setPasswordVisibility(showPlain) {
      const nextType = showPlain ? 'text' : 'password';
      passwordInput.type = nextType;
      passwordInput.setAttribute('type', nextType);
      togglePasswordBtn.textContent = showPlain ? 'Hide' : 'Show';
      togglePasswordBtn.setAttribute('aria-pressed', showPlain ? 'true' : 'false');
      if (passwordStateBadge) {
        passwordStateBadge.textContent = showPlain ? 'Visible' : 'Hidden';
        passwordStateBadge.className = showPlain
          ? 'text-[10px] sm:text-xs text-yellow-200 font-medium'
          : 'text-[10px] sm:text-xs text-slate-300/80';
      }
    }

    togglePasswordBtn.addEventListener('click', (event) =&gt; {
      event.preventDefault();
      event.stopPropagation();
      const currentlyHidden = passwordInput.getAttribute('type') === 'password' || passwordInput.type === 'password';
      setPasswordVisibility(currentlyHidden);
      passwordInput.focus();
    });

    userIdInput.addEventListener('input', clearFeedback);
    passwordInput.addEventListener('input', clearFeedback);

    loginForm.addEventListener('submit', (event) =&gt; {
      event.preventDefault();
      const idVal = userIdInput.value.trim();
      const passVal = passwordInput.value.trim();

      if (!idVal &amp;&amp; !passVal) {
        showFeedback('Please enter your ID and Password.');
        userIdInput.focus();
        return;
      }
      if (!idVal) {
        showFeedback('Please enter your ID.');
        userIdInput.focus();
        return;
      }
      if (!passVal) {
        showFeedback('Please enter your Password.');
        passwordInput.focus();
        return;
      }

      clearFeedback();
      loggedInUserSpan.textContent = idVal;
      loginForm.classList.add('hidden');
      welcomePanel.classList.remove('hidden');
      welcomePanel.classList.add('flex');
    });

    logoutBtn.addEventListener('click', () =&gt; {
      loginForm.reset();
      setPasswordVisibility(false);
      clearFeedback();
      welcomePanel.classList.add('hidden');
      welcomePanel.classList.remove('flex');
      loginForm.classList.remove('hidden');
      userIdInput.focus();
    });

    function getCleanCopiableHtml() {
      const rawHtml = document.documentElement.outerHTML;
      return '&lt;!DOCTYPE html&gt;\n' + rawHtml.replace(
        /src="data:image\/[^"]+"/,
        'src="image_d7021c.jpg"'
      );
    }

    if (codePreviewBox) {
      codePreviewBox.textContent = getCleanCopiableHtml();
    }

    if (toggleCodeViewBtn &amp;&amp; codePreviewBox) {
      toggleCodeViewBtn.addEventListener('click', () =&gt; {
        const isHidden = codePreviewBox.classList.contains('hidden');
        if (isHidden) {
          codePreviewBox.textContent = getCleanCopiableHtml();
          codePreviewBox.classList.remove('hidden');
          toggleCodeViewBtn.textContent = 'Hide Code';
        } else {
          codePreviewBox.classList.add('hidden');
          toggleCodeViewBtn.textContent = 'Show Code';
        }
      });
    }

    if (copyCodeBtn) {
      copyCodeBtn.addEventListener('click', () =&gt; {
        const codeText = getCleanCopiableHtml();
        const markCopied = () =&gt; {
          copyCodeBtn.textContent = 'Copied!';
          setTimeout(() =&gt; {
            copyCodeBtn.textContent = 'Copy HTML Code';
          }, 2000);
        };

        const fallbackCopy = () =&gt; {
          const ta = document.createElement('textarea');
          ta.value = codeText;
          ta.style.position = 'fixed';
          ta.style.opacity = '0';
          document.body.appendChild(ta);
          ta.focus();
          ta.select();
          try {
            document.execCommand('copy');
          } catch (err) {}
          document.body.removeChild(ta);
          markCopied();
        };

        if (navigator.clipboard &amp;&amp; navigator.clipboard.writeText) {
          navigator.clipboard.writeText(codeText).then(markCopied).catch(fallbackCopy);
        } else {
          fallbackCopy();
        }
      });
    }
